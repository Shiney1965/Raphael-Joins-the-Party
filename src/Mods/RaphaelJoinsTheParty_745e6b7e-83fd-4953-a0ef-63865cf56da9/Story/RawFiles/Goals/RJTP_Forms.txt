// one StackId with StackType Overwrite, so applying the new one removes the old one. Events arrive after this rule
// ends, in either order: the marker covers "old removed first", the other-status check covers "new applied first".
// Must stay ABOVE the fresh-ascend rule (that one sets DB_RJTP_Ascended).

AND
DB_RJTP_Ascended(1)
AND
DB_RJTP_AscendStatus(_Status)
AND
HasActiveStatus(_Comp, _Status, 0)
AND
IsInCombat(_Comp, 1)
AND
IsDead(_Comp, 0)
AND
NOT SpeakerGetDialog(_Comp, 0, _, _)
AND
NOT DB_RJTP_Downed(1)
AND
NOT DB_RJTP_ModdedForm(_)
THEN
DB_RJTP_AscendSwitching(1);
ApplyStatus(_Comp, _Status, -1.0, 1, _Comp);
