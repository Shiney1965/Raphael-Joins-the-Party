//REGION A. Bark engine
// Request = start the group's delay timer. The checks are repeated when the timer fires.
PROC
PROC_RJTP_BarkRequest((STRING)_Group)
AND
DB_RJTP_Recruited(1)
AND
DB_RJTP_Companion(_Comp)
AND
DB_Players(_Comp)
AND
NOT DB_RJTP_BarkCooling(_Group)
AND
NOT DB_RJTP_BarkSpent(_Group)
AND
DB_RJTP_BarkTimers(_Group, _Go, _Cd, _Delay, _Cool)
THEN
RealtimeObjectTimerLaunch(_Comp, _Go, _Delay);

IF
ObjectTimerFinished(_Comp, _Go)
AND
DB_RJTP_BarkTimers(_Group, _Go, _Cd, _Delay, _Cool)
AND
DB_RJTP_Companion((CHARACTER)_Comp)
THEN
PROC_RJTP_Bark(_Group);

// Downed is the one line a downed body may say; "Won" waits for the whole party to be out of combat.
QRY
QRY_RJTP_BarkAllowedNow((STRING)_Group)
AND
_Group == "Downed"
THEN
DB_NOOP(1);

QRY
QRY_RJTP_BarkAllowedNow((STRING)_Group)
AND
_Group == "Won"
AND
NOT DB_RJTP_Downed(1)
AND
NOT QRY_RJTP_PartyInCombat()
THEN
DB_NOOP(1);

// The ascend can still fail after DB_RJTP_Ascended(1) was set (RJTP_Forms removes the fact on StatusAttemptFailed)
QRY
QRY_RJTP_BarkAllowedNow((STRING)_Group)
AND
_Group == "Ascend"
AND
NOT DB_RJTP_Downed(1)
AND
DB_RJTP_Ascended(1)
THEN
DB_NOOP(1);

QRY
QRY_RJTP_BarkAllowedNow((STRING)_Group)
AND
_Group != "Downed"
AND
_Group != "Won"
AND
_Group != "Ascend"
AND
NOT DB_RJTP_Downed(1)
THEN
DB_NOOP(1);

// Play the group's next line. DB_RJTP_BarkStarted is set and cleared inside this one proc (two-rule marker pattern,
// as RJTP_Dialogs.txt does for the voice tests): the round-robin index is never advanced by the rule that reads it.
PROC
PROC_RJTP_Bark((STRING)_Group)
AND
DB_RJTP_Recruited(1)
AND
DB_RJTP_Companion(_Comp)
AND
DB_Players(_Comp)
AND
IsDead(_Comp, 0)
AND
NOT DB_RJTP_BarkCooling(_Group)
AND
NOT DB_RJTP_BarkSpent(_Group)
AND
NOT DB_RJTP_BarkGlobalCooling(1)
AND
QRY_RJTP_BarkAllowedNow(_Group)
AND
DB_RJTP_BarkNext(_Group, _Index)
AND
DB_RJTP_Bark(_Group, _Index, _AD)
AND
QRY_StartDialog_Fixed(_AD, _Comp)
THEN
DB_RJTP_BarkStarted(_Group, _Index);

PROC
PROC_RJTP_Bark((STRING)_Group)
AND
DB_RJTP_BarkStarted(_Group, _Index)
THEN
NOT DB_RJTP_BarkStarted(_Group, _Index);
DB_RJTP_BarkOk(1);
PROC_RJTP_BarkAdvance(_Group, _Index);
PROC_RJTP_BarkCooldown(_Group);
PROC_RJTP_BarkLog(_Group, "start accepted");

PROC
PROC_RJTP_Bark((STRING)_Group)
AND
NOT DB_RJTP_BarkOk(1)
THEN
PROC_RJTP_BarkLog(_Group, "not played (cooling, unavailable or the start was refused)");

PROC
PROC_RJTP_Bark((STRING)_Group)
AND
DB_RJTP_BarkOk(1)
THEN
NOT DB_RJTP_BarkOk(1);

PROC
PROC_RJTP_BarkAdvance((STRING)_Group, (INTEGER)_Index)
AND
DB_RJTP_BarkCount(_Group, _Count)
AND
_Index < _Count
AND
IntegerSum(_Index, 1, _Next)
THEN
NOT DB_RJTP_BarkNext(_Group, _Index);
DB_RJTP_BarkNext(_Group, _Next);

PROC
PROC_RJTP_BarkAdvance((STRING)_Group, (INTEGER)_Index)
AND
DB_RJTP_BarkCount(_Group, _Index)
THEN
NOT DB_RJTP_BarkNext(_Group, _Index);
DB_RJTP_BarkNext(_Group, 1);

PROC
PROC_RJTP_BarkCooldown((STRING)_Group)
AND
DB_RJTP_BarkTimers(_Group, _Go, _Cd, _Delay, _Cool)
AND
_Cool > 0
AND
DB_RJTP_Companion(_Comp)
THEN
DB_RJTP_BarkCooling(_Group);
RealtimeObjectTimerLaunch(_Comp, _Cd, _Cool);

PROC
PROC_RJTP_BarkCooldown((STRING)_Group)
AND
DB_RJTP_BarkTimers(_Group, _Go, _Cd, _Delay, 0)
THEN
DB_RJTP_BarkSpent(_Group);

// 8 s between any two barks, so two lines never overlap on him
PROC
PROC_RJTP_BarkCooldown((STRING)_Group)
AND
DB_RJTP_Companion(_Comp)
THEN
DB_RJTP_BarkGlobalCooling(1);
RealtimeObjectTimerLaunch(_Comp, "RJTP_BarkCD_Global", 8000);

IF
ObjectTimerFinished(_Comp, _Cd)
AND
DB_RJTP_BarkTimers(_Group, _Go, _Cd, _Delay, _Cool)
AND
DB_RJTP_BarkCooling(_Group)
THEN
NOT DB_RJTP_BarkCooling(_Group);

IF
ObjectTimerFinished(_Comp, "RJTP_BarkCD_Global")
AND
DB_RJTP_BarkGlobalCooling(1)
THEN
NOT DB_RJTP_BarkGlobalCooling(1);

// A real-time timer may not survive a load: never leave a group muted (plan review S4)
IF
SavegameLoaded()
AND
DB_RJTP_BarkCooling(_Group)
THEN
NOT DB_RJTP_BarkCooling(_Group);

IF
SavegameLoaded()
AND
DB_RJTP_BarkGlobalCooling(1)
THEN
NOT DB_RJTP_BarkGlobalCooling(1);
