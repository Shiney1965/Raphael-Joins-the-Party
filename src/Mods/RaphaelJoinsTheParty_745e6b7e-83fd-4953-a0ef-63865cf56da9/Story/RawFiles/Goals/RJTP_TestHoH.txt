//REGION Go: with the deal already made, at once; without it, only after a Yes (diff review S3)
PROC
PROC_RJTP_TestHoH((CHARACTER)_Caster)
AND
NOT DB_RJTP_TestHoHRefused(1)
AND
DB_GlobalFlag((FLAG)GLO_Monitor_State_MadeDeal_86f40f63-0bfa-4ab6-b3b0-d01475c3cbdc)
THEN
PROC_RJTP_TestHoHGo(_Caster, "RJTP_Msg_TestHoH_Go");

PROC
PROC_RJTP_TestHoH((CHARACTER)_Caster)
AND
NOT DB_RJTP_TestHoHRefused(1)
AND
NOT DB_GlobalFlag((FLAG)GLO_Monitor_State_MadeDeal_86f40f63-0bfa-4ab6-b3b0-d01475c3cbdc)
THEN
PROC_RJTP_Box(_Caster, "RJTP_Msg_TestHoH_ConfirmDeal", 1);

// The deal exactly as vanilla's own debug shortcut sets it (Act3_WYR_RaphaelTango.txt:752-759)
PROC
PROC_RJTP_BoxAnswered((CHARACTER)_Player, "RJTP_Msg_TestHoH_ConfirmDeal", 1)
AND
NOT DB_GlobalFlag((FLAG)GLO_Monitor_State_MadeDeal_86f40f63-0bfa-4ab6-b3b0-d01475c3cbdc)
AND
DB_CurrentLevel("CTY_Main_A")
THEN
DB_RJTP_TestHoHDealGiven(1);
PROC_GlobalSetFlagAndCache((FLAG)GLO_Monitor_State_MadeDeal_86f40f63-0bfa-4ab6-b3b0-d01475c3cbdc);
PROC_GlobalSetFlagAndCache((FLAG)WYR_RaphaelTango_State_DiscussedDeal_9896c94b-431e-4b0e-90ce-80293ffb1198);
ToInventory((ITEM)S_GLO_OrphicHammer_9013ef0d-c6ee-4137-b17a-c1934079e10d, _Player, 1, 1, 1);
PROC_RJTP_TestHoHGo(_Player, "RJTP_Msg_TestHoH_GoDeal");

// "No" (or Escape): nothing is changed
PROC
PROC_RJTP_BoxAnswered((CHARACTER)_Player, "RJTP_Msg_TestHoH_ConfirmDeal", 0)
THEN
DB_NOOP(1);

PROC
PROC_RJTP_TestHoHGo((CHARACTER)_Player, (STRING)_Key)
THEN
PROC_TeleportPartiesTo((TRIGGER)S_LOW_HouseOfHopeEntrance_f39ce96a-e495-49dd-a171-b3b160a43d80, "");
PROC_RJTP_Box(_Player, _Key, 0);

// Diff review M1: vanilla sets NoHammerInHouse for every party member who enters without the hammer
// (HoH2:724-731); the test deal gives it to one character only, so keep the flag clear in a test-deal save
IF
FlagSet((FLAG)LOW_HouseOfHope_State_NoHammerInHouse_29adfa78-2141-4599-a46a-53dd49acdb23, _, _)
AND
DB_RJTP_TestHoHDealGiven(1)
THEN
ClearFlag((FLAG)LOW_HouseOfHope_State_NoHammerInHouse_29adfa78-2141-4599-a46a-53dd49acdb23);
