//REGION Message layer (review 08 S1: resolve the loca key first, as WW_AIJP does; fall back to the raw key)
PROC
PROC_RJTP_Box((CHARACTER)_Player, (STRING)_Key, (INTEGER)_YesNo)
AND
ResolveTranslatedString(_Key, _Text)
AND
_Text != ""
AND
DB_RJTP_BoxText(_Key, _Old)
AND
_Old != _Text
THEN
NOT DB_RJTP_BoxText(_Key, _Old);

PROC
PROC_RJTP_Box((CHARACTER)_Player, (STRING)_Key, (INTEGER)_YesNo)
AND
ResolveTranslatedString(_Key, _Text)
AND
_Text != ""
AND
NOT DB_RJTP_BoxText(_Key, _Text)
THEN
DB_RJTP_BoxText(_Key, _Text);

PROC
PROC_RJTP_Box((CHARACTER)_Player, (STRING)_Key, (INTEGER)_YesNo)
AND
NOT DB_RJTP_BoxText(_Key, _)
THEN
DB_RJTP_BoxText(_Key, _Key);

PROC
PROC_RJTP_Box((CHARACTER)_Player, (STRING)_Key, 1)
AND
DB_RJTP_Step(_OldStep)
THEN
NOT DB_RJTP_Step(_OldStep);

// v0.3e CRASH FIX (2026-10-15, Osiris log of the author's crash): a Yes/No box opened with the RESOLVED text (346 bytes)
// came back as MessageBoxYesNoClosed(_, text, 1936485473) - the answer was text bytes 92-95 ("alls" of "halls"),
// and the game crashed at once. The engine overruns a fixed buffer with a long message. Vanilla only ever passes a
// short loca KEY to OpenMessageBoxYesNo and matches the answer on that key (Act1_PLA_TavernInvestigation_Surroundings
// .txt:626, :655), so Yes/No boxes now do exactly that. Plain boxes (no answer value) keep the resolved text.
PROC
PROC_RJTP_Box((CHARACTER)_Player, (STRING)_Key, 1)
THEN
DB_RJTP_Step(_Key);
OpenMessageBoxYesNo(_Player, _Key);

PROC
PROC_RJTP_Box((CHARACTER)_Player, "RJTP_Msg_ItemGone", 0)
AND
DB_RJTP_Step(_OldStep)
THEN
NOT DB_RJTP_Step(_OldStep);

PROC
PROC_RJTP_Box((CHARACTER)_Player, "RJTP_Msg_ItemGone", 0)
THEN
DB_RJTP_Step("RJTP_Msg_ItemGone");

// v0.3e (crash-fix review SF1): plain boxes pass the key too, exactly as vanilla does (Act3b_LOW_HouseOfHope2.txt:3643
// OpenMessageBox(_Player, "LOW_HouseOfHope_GatherParty")); the text comes from Localization/RJTP_MessageKeys
PROC
PROC_RJTP_Box((CHARACTER)_Player, (STRING)_Key, 0)
THEN
OpenMessageBox(_Player, _Key);

IF
MessageBoxYesNoClosed(_Player, _Key, _Result)
AND
DB_RJTP_Step(_Key)
AND
_Key != "RJTP_Msg_ItemGone"
AND
_Result >= 0
AND
_Result <= 1
THEN
NOT DB_RJTP_Step(_Key);
PROC_RJTP_BoxAnswered(_Player, _Key, _Result);

// Never act on an unreadable answer (the v0.3d crash delivered 1936485473): record it, close the step, cancel any
// open offer (the next portal use offers again) and say so
IF
MessageBoxYesNoClosed(_Player, _Key, _Result)
AND
DB_RJTP_Step(_Key)
AND
_Key != "RJTP_Msg_ItemGone"
AND
_Result > 1
THEN
NOT DB_RJTP_Step(_Key);
DB_RJTP_BadBoxResult(_Key, _Result);
PROC_RJTP_BadAnswer(_Player);

IF
MessageBoxYesNoClosed(_Player, _Key, _Result)
AND
DB_RJTP_Step(_Key)
AND
_Key != "RJTP_Msg_ItemGone"
AND
_Result < 0
THEN
NOT DB_RJTP_Step(_Key);
DB_RJTP_BadBoxResult(_Key, _Result);
PROC_RJTP_BadAnswer(_Player);

// Only the plain OK box (ItemGone) is answered by a plain close; a Yes/No box never is (review 10 SF-A)
IF
MessageBoxClosed(_Player, "RJTP_Msg_ItemGone")
AND
DB_RJTP_Step("RJTP_Msg_ItemGone")
THEN
NOT DB_RJTP_Step("RJTP_Msg_ItemGone");
PROC_RJTP_BoxClosed(_Player, "RJTP_Msg_ItemGone");

// An unreadable answer: close an open offer (only then; review MINOR 3), and say so
PROC
PROC_RJTP_BadAnswer((CHARACTER)_Player)
AND
DB_RJTP_OfferInProgress(1)
THEN
PROC_RJTP_CancelOffer();

PROC
PROC_RJTP_BadAnswer((CHARACTER)_Player)
THEN
PROC_RJTP_Box(_Player, "RJTP_Msg_BadAnswer", 0);

// Catch-all definitions so every key has a handler
PROC
PROC_RJTP_BoxAnswered((CHARACTER)_Player, (STRING)_Key, (INTEGER)_Result)
THEN
DB_NOOP(1);

PROC
PROC_RJTP_BoxClosed((CHARACTER)_Player, (STRING)_Key)
THEN
DB_NOOP(1);

PROC
PROC_RJTP_TellAvatars((STRING)_Key)
AND
DB_Avatars(_Avatar)
THEN
