// Raphael Joins the Party - v0.3c TEST: the private scene of his camp conversation (proposal 5.7), fade to black.
// The dialog (RJTP_CAMP_Raphael, branch "You've been watching me all evening.") ends on a line whose node sets the
// Object flag RJTP_Event_Private on the companion. This goal then fades the screen to black, fades it back a few
// seconds later, marks the companion with RJTP_State_Romanced (unlocks one camp topic) and plays one overhead line.
// No vanilla romance flag, approval or relationship state is read or written.
// A NEW goal on purpose: the INIT facts of a goal already in a save are never re-run (notes in RJTP_Forms.txt,
// RJTP_Companion.txt and RJTP_Cure_v02.txt).
// Dialog and flag resources carry GUIDs from src/ids.json, so these facts are written by the generator.
// BEGIN GENERATED by tools/gen_dialogs.py - do not edit by hand
DB_RJTP_PrivateFlag((FLAG)RJTP_Event_Private_91374685-6138-4d76-bd5f-cd474eae4317);
DB_RJTP_PrivateDoneFlag((FLAG)RJTP_State_Romanced_90b79f90-0ffd-4704-b75b-fb94b53bf279);
DB_RJTP_PrivateAfterAD((DIALOGRESOURCE)RJTP_AD_PrivateAfter_6f333068-90cd-4715-b11b-d910d3a20a8d);

//REGION Private scene: fade out on the dialog's last line, fade in, mark, one overhead line
// DB_RJTP_PrivateRunning(1) spans events on purpose; it is cleared by the last timer and on load.
// v0.3b never faded: its flag files said Usage 5 (= a GLOBAL flag; vanilla object flags are Usage 4), so the game
// raised FlagSet(flag, NULL, instance) and a rule keyed on the speaker never matched (Osiris log, 2026-10-08).
// The flags are Usage 4 now, and this rule accepts the event whoever the "speaker" argument is.
IF
FlagSet(_Flag, _, _)
AND
DB_RJTP_PrivateFlag(_Flag)
AND
DB_RJTP_Companion(_Comp)
THEN
ClearFlag(_Flag, _Comp);
ClearFlag(_Flag, NULL_00000000-0000-0000-0000-000000000000, 0);
PROC_RJTP_PrivateBegin(_Comp);

PROC
PROC_RJTP_PrivateBegin((CHARACTER)_Comp)
AND
NOT DB_RJTP_PrivateRunning(1)
THEN
DB_RJTP_PrivateRunning(1);
DB_RJTP_PrivateWaitEnd(1);
PROC_RJTP_PrivateMark(_Comp);
RealtimeObjectTimerLaunch(_Comp, "RJTP_PrivateFadeOut", 5500);
RealtimeObjectTimerLaunch(_Comp, "RJTP_PrivateFadeIn", 20000);

// The flag is raised when the scene's last line STARTS (Osiris log 2026-10-08), and that line is 8 s long: the
// fade to black begins 5.5 s into it. The fade back is counted from DialogEnded (vanilla starts its own
// fade-to-black scenes from DialogEnded too); the 20 s timer above is only a fallback if that event never comes.
IF
ObjectTimerFinished(_Comp, "RJTP_PrivateFadeOut")
AND
DB_RJTP_Companion((CHARACTER)_Comp)
AND
DB_RJTP_PrivateRunning(1)
THEN
PROC_RJTP_PrivateFadeOut();

// The player skipped the last line (the dialog ended before the fade began): go to black now. Then 4.5 s of black.
IF
DialogEnded((DIALOGRESOURCE)RJTP_CAMP_Raphael_b8de429b-ac5c-46e5-af00-822da8d57f97, _)
AND
DB_RJTP_PrivateWaitEnd(1)
AND
DB_RJTP_Companion(_Comp)
THEN
NOT DB_RJTP_PrivateWaitEnd(1);
RealtimeObjectTimerCancel(_Comp, "RJTP_PrivateFadeOut");
RealtimeObjectTimerCancel(_Comp, "RJTP_PrivateFadeIn");
PROC_RJTP_PrivateFadeOut();
RealtimeObjectTimerLaunch(_Comp, "RJTP_PrivateFadeIn", 4500);

// Style 0 = black (vanilla GLO_Camp / GOB_WarlockProposal use the same call). One fade per avatar = per player.
PROC
PROC_RJTP_PrivateFadeOut()
AND
DB_Avatars(_Avatar)
AND
NOT DB_RJTP_PrivateFaded(_Avatar)
THEN
ScreenFadeTo(_Avatar, 2.0, 0, "RJTP_Private");
DB_RJTP_PrivateFaded(_Avatar);
RealtimeObjectTimerLaunch(_Avatar, "RJTP_PrivateFadeSafety", 30000);

PROC
PROC_RJTP_PrivateFadeIn()
AND
DB_RJTP_PrivateFaded(_Avatar)
THEN
ClearScreenFade(_Avatar, 2.0, "RJTP_Private");
NOT DB_RJTP_PrivateFaded(_Avatar);
RealtimeObjectTimerCancel(_Avatar, "RJTP_PrivateFadeSafety");

// Safety net (the shape of vanilla GLO_Camp's "GLO_Camp_Fade_EmergencyTimeout"): this timer sits on the faded
// character, so the screen comes back even if the companion's own timer never fires.
IF
ObjectTimerFinished(_Avatar, "RJTP_PrivateFadeSafety")
AND
DB_RJTP_PrivateFaded((CHARACTER)_Avatar)
THEN
ClearScreenFade((CHARACTER)_Avatar, 2.0, "RJTP_Private");
NOT DB_RJTP_PrivateFaded((CHARACTER)_Avatar);
NOT DB_RJTP_PrivateRunning(1);

IF
ObjectTimerFinished(_Comp, "RJTP_PrivateFadeIn")
AND
DB_RJTP_Companion((CHARACTER)_Comp)
THEN
PROC_RJTP_PrivateFadeIn();
RealtimeObjectTimerLaunch(_Comp, "RJTP_PrivateAfter", 3500);

PROC
PROC_RJTP_PrivateMark((CHARACTER)_Comp)
AND
DB_RJTP_PrivateDoneFlag(_Flag)
THEN
SetFlag(_Flag, _Comp);

// The overhead line after the scene (an automated dialog, the shape the barks use). A refused start is harmless.
IF
ObjectTimerFinished(_Comp, "RJTP_PrivateAfter")
AND
DB_RJTP_Companion((CHARACTER)_Comp)
AND
DB_RJTP_PrivateRunning(1)
AND
IsDead(_Comp, 0)
AND
DB_RJTP_PrivateAfterAD(_AD)
AND
QRY_StartDialog_Fixed(_AD, _Comp)
THEN
DB_NOOP(1);

IF
ObjectTimerFinished(_Comp, "RJTP_PrivateAfter")
AND
DB_RJTP_Companion((CHARACTER)_Comp)
AND
DB_RJTP_PrivateRunning(1)
THEN
NOT DB_RJTP_PrivateRunning(1);

// Never leave the screen black: a load, or the uninstall step, clears any fade this goal started.
// A client-facing call at SavegameLoaded can be too early (RJTP_AAA_Init defers its message box for the same
// reason), so the fade is cleared now and once more when gameplay has started.
IF
SavegameLoaded()
AND
DB_RJTP_PrivateFaded(_Avatar)
THEN
DB_RJTP_PrivateClearPending(_Avatar);
ClearScreenFade(_Avatar, 0.5, "RJTP_Private");
NOT DB_RJTP_PrivateFaded(_Avatar);

IF
LevelGameplayStarted(_, _)
AND
DB_RJTP_PrivateClearPending(_Avatar)
THEN
NOT DB_RJTP_PrivateClearPending(_Avatar);
ClearScreenFade(_Avatar, 0.5, "RJTP_Private");

// A stale event flag would never raise FlagSet again (v0.3b left it set as a global flag)
IF
SavegameLoaded()
AND
DB_RJTP_PrivateFlag(_Flag)
AND
DB_RJTP_Companion(_Comp)
THEN
ClearFlag(_Flag, _Comp);
ClearFlag(_Flag, NULL_00000000-0000-0000-0000-000000000000, 0);

IF
SavegameLoaded()
AND
DB_RJTP_PrivateRunning(1)
THEN
NOT DB_RJTP_PrivateRunning(1);

IF
SavegameLoaded()
AND
DB_RJTP_PrivateWaitEnd(1)
THEN
NOT DB_RJTP_PrivateWaitEnd(1);

PROC
PROC_RJTP_Uninstall((CHARACTER)_Caster)
AND
DB_RJTP_PrivateFaded(_Avatar)
THEN
ClearScreenFade(_Avatar, 0.5, "RJTP_Private");
NOT DB_RJTP_PrivateFaded(_Avatar);
RealtimeObjectTimerCancel(_Avatar, "RJTP_PrivateFadeSafety");

PROC
PROC_RJTP_Uninstall((CHARACTER)_Caster)
AND
DB_RJTP_Companion(_Comp)
THEN
RealtimeObjectTimerCancel(_Comp, "RJTP_PrivateFadeOut");
RealtimeObjectTimerCancel(_Comp, "RJTP_PrivateFadeIn");
RealtimeObjectTimerCancel(_Comp, "RJTP_PrivateAfter");

PROC
PROC_RJTP_Uninstall((CHARACTER)_Caster)
AND
DB_RJTP_PrivateWaitEnd(1)
THEN
NOT DB_RJTP_PrivateWaitEnd(1);

PROC
PROC_RJTP_Uninstall((CHARACTER)_Caster)
AND
DB_RJTP_PrivateRunning(1)
THEN
NOT DB_RJTP_PrivateRunning(1);
//END_REGION

//REGION Private scene: Haarlep's recorded lines (same actor as Raphael, but stored under Haarlep's speaker id)
// The scene's first line (Raphael's own) raises RJTP_Event_VoiceHaarlep. 0.8 s later, while that line is playing,
// the SE layer switches the companion's speaker id to Haarlep's (PROC_RJTP_VoiceRequest is a no-op proc that
// RJTP_Server.lua observes), so the Haarlep lines that follow are voiced. Every exit from the scene ends the
// dialog, and DialogEnded switches him back. Without the Script Extender those lines are subtitle-only.
// The flag GUID is a literal here (not an INIT fact): this goal is already in saves made with v0.3b, and the INIT
// facts of an existing goal are never re-run. The build's goal-constant check verifies the GUID.
PROC
PROC_RJTP_VoiceRequest((STRING)_Who)
THEN
DB_NOOP(1);

IF
FlagSet((FLAG)RJTP_Event_VoiceHaarlep_1b635df9-1131-4164-8535-d03683220fdf, _, _)
AND
DB_RJTP_Companion(_Comp)
THEN
ClearFlag((FLAG)RJTP_Event_VoiceHaarlep_1b635df9-1131-4164-8535-d03683220fdf, _Comp);
ClearFlag((FLAG)RJTP_Event_VoiceHaarlep_1b635df9-1131-4164-8535-d03683220fdf, NULL_00000000-0000-0000-0000-000000000000, 0);
DB_RJTP_PrivateVoice(1);
RealtimeObjectTimerLaunch(_Comp, "RJTP_PrivateVoice", 800);

IF
ObjectTimerFinished(_Comp, "RJTP_PrivateVoice")
AND
DB_RJTP_Companion((CHARACTER)_Comp)
AND
DB_RJTP_PrivateVoice(1)
THEN
PROC_RJTP_VoiceRequest("haarlep");

// Back to his own voice when the dialog ends, however it ends. The marker also stops a late timer from switching
// him after the dialog is over.
IF
DialogEnded((DIALOGRESOURCE)RJTP_CAMP_Raphael_b8de429b-ac5c-46e5-af00-822da8d57f97, _)
AND
DB_RJTP_PrivateVoice(1)
AND
DB_RJTP_Companion(_Comp)
THEN
NOT DB_RJTP_PrivateVoice(1);
RealtimeObjectTimerCancel(_Comp, "RJTP_PrivateVoice");
PROC_RJTP_VoiceRequest("raphael");

// A load resets his speaker id anyway (RJTP_Server.lua, FixVoice two seconds after SavegameLoaded)
IF
SavegameLoaded()
AND
DB_RJTP_PrivateVoice(1)
THEN
NOT DB_RJTP_PrivateVoice(1);

PROC
PROC_RJTP_Uninstall((CHARACTER)_Caster)
AND
DB_RJTP_PrivateVoice(1)
THEN
NOT DB_RJTP_PrivateVoice(1);
PROC_RJTP_VoiceRequest("raphael");
//END_REGION
